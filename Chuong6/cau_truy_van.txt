CREATE TABLE TONKHO (
    id_sp VARCHAR(4) PRIMARY KEY,
    ten_sp VARCHAR(40),
    id_ncc VARCHAR(2),
    nam_sx INT,
    don_gia INT,
    so_luong INT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
